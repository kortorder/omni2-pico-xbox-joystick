cmake_minimum_required(VERSION 3.20)
# The shared pid.codes ID is not unique and must stay within private tests.
# Do not ship, redistribute, sell or manufacture hardware using it.
option(OMNI_USE_PRIVATE_TEST_ID "Explicitly use shared 1209:0002 only in private bench testing" OFF)
set(OMNI_USB_VID "" CACHE STRING "Assigned USB vendor ID, in hexadecimal")
set(OMNI_USB_PID "" CACHE STRING "Assigned USB product ID, in hexadecimal")
if(OMNI_USE_PRIVATE_TEST_ID)
    if(NOT "${OMNI_USB_VID}" STREQUAL "" OR NOT "${OMNI_USB_PID}" STREQUAL "")
        message(FATAL_ERROR "Select the private test ID or provide your own assigned IDs, not both.")
    endif()
    set(omni_vid 0x1209)
    set(omni_pid 0x0002)
else()
    foreach(id IN ITEMS OMNI_USB_VID OMNI_USB_PID)
        if(NOT "${${id}}" MATCHES "^0[xX][0-9a-fA-F][0-9a-fA-F][0-9a-fA-F][0-9a-fA-F]$")
            message(FATAL_ERROR "Supply your own OMNI_USB_VID and OMNI_USB_PID as 0xHHHH, or explicitly select OMNI_USE_PRIVATE_TEST_ID for a private bench build. See README.md.")
        endif()
        math(EXPR number "${${id}}")
        if(number LESS 1 OR number GREATER 65534)
            message(FATAL_ERROR "Placeholder USB identities are not supported.")
        endif()
    endforeach()
    math(EXPR vid_num "${OMNI_USB_VID}")
    math(EXPR pid_num "${OMNI_USB_PID}")
    if(vid_num EQUAL 4617 AND pid_num EQUAL 2)
        message(FATAL_ERROR "1209:0002 is the shared private test ID. Select OMNI_USE_PRIVATE_TEST_ID explicitly.")
    endif()
    set(omni_vid "${OMNI_USB_VID}")
    set(omni_pid "${OMNI_USB_PID}")
endif()
if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
    message(FATAL_ERROR "Set PICO_SDK_PATH to an external Raspberry Pi Pico SDK 2.3.1 checkout.")
endif()
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(omni_right_stick C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
if(NOT PICO_SDK_VERSION_STRING STREQUAL "2.3.1")
    message(FATAL_ERROR "This source was tested with Pico SDK 2.3.1. Use the pinned SDK and its matching submodules.")
endif()
if(NOT PICO_BOARD STREQUAL "pico_w")
    message(FATAL_ERROR "Select -DPICO_BOARD=pico_w for the RP2040 Pico W/WH.")
endif()
if(NOT TARGET pico_btstack_classic)
    message(FATAL_ERROR "Initialize the Pico SDK's pinned lib/btstack submodule.")
endif()
add_executable(omni_right_stick src/main.cpp src/pair_button.cpp
    src/omni_bluetooth.cpp src/omni_mouse_report.cpp
    src/omni_stick.cpp src/omni_usb.cpp src/omni_maintenance.cpp
    src/omni_nudges.cpp src/omni_nudge_settings.cpp)
target_include_directories(omni_right_stick PRIVATE src)
option(OMNI_CAPTURE_SHORT_KEY "Strict, wired short-key observation for private commissioning" OFF)
set(OMNI_APPROVED_MOUSE2_FILE "" CACHE FILEPATH "Private owner-only file with six verified raw Mouse 2 address bytes")
set(omni_variant_description "Omni2 Mouse 2 host - strict 16-byte minimum")
if(OMNI_CAPTURE_SHORT_KEY AND OMNI_APPROVED_MOUSE2_FILE)
    message(FATAL_ERROR "Strict observation and private compatibility are mutually exclusive.")
elseif(OMNI_CAPTURE_SHORT_KEY)
    target_compile_definitions(omni_right_stick PRIVATE OMNI_CAPTURE_SHORT_KEY=1)
    set(omni_variant_description "Omni2 Mouse 2 host - strict 16-byte wired observation")
elseif(OMNI_APPROVED_MOUSE2_FILE)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(omni_pin_header "${CMAKE_CURRENT_BINARY_DIR}/private-peer/omni_approved_peer.hpp")
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/cmake/private_peer.py"
        "${OMNI_APPROVED_MOUSE2_FILE}" "${omni_pin_header}" "${CMAKE_CURRENT_LIST_DIR}"
        RESULT_VARIABLE omni_pin_status)
    if(NOT omni_pin_status EQUAL 0)
        message(FATAL_ERROR "Private peer pin was not accepted; no compatibility image can be built.")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${OMNI_APPROVED_MOUSE2_FILE}"
        "${CMAKE_CURRENT_LIST_DIR}/cmake/private_peer.py")
    target_include_directories(omni_right_stick PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/private-peer")
    target_compile_definitions(omni_right_stick PRIVATE OMNI_SHORT_KEY_COMPAT=1
        ENABLE_MUTUAL_AUTHENTICATION_FOR_LEGACY_SECURE_CONNECTIONS=1)
    set(omni_variant_description "Omni2 Mouse 2 host - private single-peer legacy key compatibility")
endif()
# pid.codes 1209:0002 is a shared private-test ID, not unique or for distribution.
target_compile_definitions(omni_right_stick PRIVATE
    OMNI_USB_VID=${omni_vid} OMNI_USB_PID=${omni_pid} OMNI_USB_POWER_MA=500 CYW43_LWIP=0)
target_link_libraries(omni_right_stick PRIVATE pico_stdlib tinyusb_device
    pico_cyw43_arch_poll pico_btstack_classic pico_btstack_cyw43 hardware_watchdog)
pico_enable_stdio_usb(omni_right_stick 0)
pico_enable_stdio_uart(omni_right_stick 0)
pico_set_program_name(omni_right_stick "Omni Right Stick")
pico_set_program_description(omni_right_stick "${omni_variant_description}")
pico_set_program_version(omni_right_stick "0.1.5")
pico_add_extra_outputs(omni_right_stick)
